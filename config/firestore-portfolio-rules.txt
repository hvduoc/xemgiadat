// Firestore Security Rules cho Portfolio BDS
// Thêm vào file firestore.rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Portfolio collection - Chỉ owner mới được đọc/ghi portfolio riêng tư
    match /portfolios/{portfolioId} {
      // Cho phép đọc nếu:
      // - Là owner của portfolio
      // - Hoặc portfolio có visibility = 'public'
      allow read: if request.auth != null && 
        (resource.data.userId == request.auth.uid || 
         resource.data.visibility == 'public');
      
      // Chỉ owner mới được tạo, sửa, xóa
      allow create, update: if request.auth != null && 
        request.resource.data.userId == request.auth.uid;
      
      allow delete: if request.auth != null && 
        resource.data.userId == request.auth.uid;
      
      // Validate dữ liệu khi tạo/cập nhật
      allow create, update: if request.auth != null &&
        request.auth.uid == request.resource.data.userId &&
        request.resource.data.keys().hasAll(['name', 'visibility', 'userId']) &&
        request.resource.data.visibility in ['private', 'public'] &&
        request.resource.data.name is string &&
        request.resource.data.name.size() > 0 &&
        request.resource.data.name.size() <= 200;
    }
    
    // Existing rules cho listings, users, feedback...
    // ... keep existing rules ...
  }
}

// Notes:
// - Mỗi user chỉ có thể đọc/ghi portfolio của chính mình
// - Portfolio public có thể được đọc bởi tất cả user đã đăng nhập
// - Validate tên portfolio (bắt buộc, tối đa 200 ký tự)
// - Validate visibility chỉ nhận 'private' hoặc 'public'
