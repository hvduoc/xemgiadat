// Copy toàn bộ nội dung này vào Firebase Console → Firestore → Rules
// Sau đó click "Publish" để áp dụng

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // --- QUY TẮC CHO TIN ĐĂNG (LISTINGS) ---
    match /listings/{listingId} {
      
      // Bất kỳ ai cũng có thể đọc được tin đăng
      allow read: if true;

      // Chỉ người dùng đã đăng nhập mới được tạo tin
      allow create: if request.auth != null
              && request.resource.data.name is string
              && request.resource.data.name.size() > 0
              && request.resource.data.priceValue is number
              && request.resource.data.priceValue > 0
              && request.resource.data.userId == request.auth.uid; // Đảm bảo userId match với auth
      
      // --- PHẦN SỬA LỖI QUAN TRỌNG ---
      // Cho phép CẬP NHẬT hoặc XÓA nếu:
      // 1. Người yêu cầu là chủ sở hữu của tin đăng
      // HOẶC (||)
      // 2. Người yêu cầu là Admin (dựa trên User ID của bạn)
      allow update, delete: if request.auth.uid == resource.data.userId 
                              || request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1';
    }

    // --- QUY TẮC CHO HỒ SƠ NGƯỜI DÙNG (USERS) ---
    match /users/{userId} {
      // Bất kỳ ai cũng có thể đọc hồ sơ (để xem avatar, tên)
      allow read: if true;

      // Chỉ chủ sở hữu của hồ sơ mới được phép ghi (tạo, cập nhật)
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // --- QUY TẮC CHO VÍ BẤT ĐỘNG SẢN (PORTFOLIOS) ---
    match /portfolios/{portfolioId} {
      // Cho phép đọc nếu:
      // - Là owner của portfolio
      // - Hoặc portfolio có visibility = 'public'
      // - Hoặc là Admin
      allow read: if request.auth != null && 
        (resource.data.userId == request.auth.uid || 
         resource.data.visibility == 'public' ||
         request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1');

      // Chỉ owner mới được tạo, sửa
      allow create, update: if request.auth != null && 
        request.resource.data.userId == request.auth.uid;

      // Chỉ owner hoặc admin mới được xóa
      allow delete: if request.auth != null && 
        (resource.data.userId == request.auth.uid ||
         request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1');

      // Validate dữ liệu khi tạo/cập nhật
      allow create, update: if request.auth != null &&
        request.auth.uid == request.resource.data.userId &&
        request.resource.data.keys().hasAll(['name', 'visibility', 'userId']) &&
        request.resource.data.visibility in ['private', 'public'] &&
        request.resource.data.name is string &&
        request.resource.data.name.size() > 0 &&
        request.resource.data.name.size() <= 200 &&
        // Validate field types
        (request.resource.data.price == null || request.resource.data.price is string) &&
        (request.resource.data.area == null || request.resource.data.area is string) &&
        (request.resource.data.notes == null || request.resource.data.notes is string) &&
        (request.resource.data.images == null || request.resource.data.images is list) &&
        (request.resource.data.lat == null || request.resource.data.lat is number) &&
        (request.resource.data.lng == null || request.resource.data.lng is number);
    }

    // --- QUY TẮC CHO FEEDBACK/CONTACT (nếu có) ---
    match /feedback/{feedbackId} {
      // Chỉ admin được đọc feedback
      allow read: if request.auth != null && 
        request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1';
      
      // Ai cũng có thể gửi feedback (không cần đăng nhập)
      allow create: if true;
      
      // Chỉ admin được xóa/sửa feedback
      allow update, delete: if request.auth != null && 
        request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1';
    }

    // --- QUY TẮC CHO ANALYTICS/LOGS (nếu có) ---
    match /analytics/{analyticsId} {
      // Chỉ admin được đọc/ghi analytics
      allow read, write: if request.auth != null && 
        request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1';
    }

    // --- QUY TẮC MẶC ĐỊNH - DENY ALL ---
    // Từ chối truy cập vào tất cả collections khác
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
