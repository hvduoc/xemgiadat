      // Cho phép đọc ảnh nếu user đã đăng nhập

      // Cho phép upload nếu user đã đăng nhập và là owner

         request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1');
      
      // Cho phép xóa nếu user đã đăng nhập và là owner  
      allow delete: if request.auth != null && 
        (request.auth.uid == userId ||
         request.auth.uid == 'FEpPWWT1EaTWQ9FOqBxWN5FeEJk1');
    }
    
    // FALLBACK RULE - DENY ALL OTHER ACCESS
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
